    typeof navigator !== 'undefined' &&
    /Mac|iPhone|iPad/.test(navigator.platform)
      ? '⌘ Enter'
      : 'Ctrl Enter';

  readonly needing = computed(() =>
    this.products().filter(
      (product) => product.onHand <= 0 || product.suggest > 0
    )
  );

  readonly suggestedProducts = computed(() =>
    this.products().filter((product) => product.suggest > 0)
  );

  readonly suggestedUnits = computed(() =>
    this.suggestedProducts().reduce(
      (sum, product) => sum + product.suggest,
      0
    )
  );

  readonly filtered = computed(() => {
    const query = this.search().trim().toLowerCase();

    return this.products()
      .filter(
        (product) =>
          !this.onlyNeeded() ||
          product.onHand <= 0 ||
          product.suggest > 0
      )
      .filter(
        (product) =>
          !query ||
          product.name.toLowerCase().includes(query) ||
        product.sku.toLowerCase().includes(query) ||
        (product.barcode ?? '').toLowerCase().includes(query) ||
        (product.category ?? '').toLowerCase().includes(query)
      );
  });

  readonly lines = computed<BasketLine[]>(() => {
    const cart = this.cart();

    return this.products()
      .filter((product) => (cart[product.id] ?? 0) > 0)
      .map((product) => {
        const qty = cart[product.id];
        const after =
          Math.max(0, product.onHand) +
          product.incoming +
          qty;

        return {
          ...product,
          qty,
          after,
          daysAfter:
            product.dailyRate > 0
              ? after / product.dailyRate
              : null,
        };
      });
  });

  readonly totalUnits = computed(() =>
    this.lines().reduce((sum, line) => sum + line.qty, 0)
  );

  async ngOnInit(): Promise<void> {
    const { data: suppliers, error: suppliersError } = await supabase
      .from('suppliers')
      .select('id, name, order_email')
      .eq('active', true)
      .order('name');

    if (suppliersError) {
      this.error.set(suppliersError.message);
    } else {
      this.suppliers.set(
        (suppliers ?? []) as OrderSupplierOption[]
      );
      this.selectedSupplierId.set(null);
    }

    const queryParams = this.route.snapshot.queryParamMap;
    const fromObject = queryParams.get('object');
    const fromProduct = queryParams.get('product');
    const fromQty = Number(queryParams.get('qty')) || 0;

    if (this.isAdmin()) {
      const { data } = await supabase
        .from('objects')
        .select('*')
        .eq('active', true)
        .order('name');

      this.objects.set((data ?? []) as BusinessObject[]);

      const chosen =
        this.objects().find((object) => object.id === fromObject)?.id ??
        this.objects()[0]?.id ??
        null;

      this.selectedObjectId.set(chosen);
    } else {
      const ownObjectId = this.auth.profile()?.object_id ?? null;
      this.selectedObjectId.set(ownObjectId);

      if (ownObjectId) {
        const { data } = await supabase
          .from('objects')
          .select('*')
          .eq('id', ownObjectId)
          .maybeSingle();

        if (data) {
          this.objects.set([data as BusinessObject]);
        }
      }
    }

    const objectId = this.selectedObjectId();

    if (objectId) {
      await this.loadProducts(objectId);
    }

    if (fromProduct) {
      const product = this.products().find(
        (item) => item.id === fromProduct
      );

      if (product) {
        this.cart.set({
          [product.id]:
            fromQty > 0
              ? fromQty
              : Math.max(1, product.suggest),
        });
      }
    }

    this.loading.set(false);
  }

  async onObjectChange(objectId: string): Promise<void> {
    if (!objectId) return;

    this.selectedObjectId.set(objectId);
    this.cart.set({});
    this.search.set('');
    this.scanFeedback.set(null);
    this.onlyNeeded.set(false);

    void this.router.navigate([], {

  private async loadProducts(objectId: string): Promise<void> {
    this.productsLoading.set(true);
    this.error.set(null);

    const since = new Date();
    since.setHours(0, 0, 0, 0);
    since.setDate(since.getDate() - (WINDOW_DAYS - 1));

    const [assigned, inventory, sales, openOrders] = await Promise.all([
      supabase
        .from('object_products')
        .select(`
          product:products (
            id,
            sku,
            barcode,
            name,
            category,
            active
          )
        `)
        .eq('object_id', objectId),

      supabase
        .from('inventory')
        .select('product_id, quantity')
        .eq('object_id', objectId),

      supabase
        .from('stock_movements')
        .select('product_id, quantity')
        .eq('object_id', objectId)
        .eq('event', 'SALE')
        .gte('created_at', since.toISOString())
        .limit(5000),

      supabase
        .from('orders')
        .select(`
          status,
          order_items (
            product_id,
            quantity
          )
        `)
        .eq('object_id', objectId)
        .in('status', ['PENDING', 'APPROVED']),
    ]);

    if (assigned.error) {
      this.error.set(assigned.error.message);
      this.products.set([]);
      this.productsLoading.set(false);
      return;
    }

    if (inventory.error) {
      this.error.set(inventory.error.message);
      this.products.set([]);
      this.productsLoading.set(false);
      return;
    }

    if (sales.error) {
      this.error.set(sales.error.message);
      this.products.set([]);
      this.productsLoading.set(false);
      return;
    }

    if (openOrders.error) {
      this.error.set(openOrders.error.message);
      this.products.set([]);
      this.productsLoading.set(false);
      return;
    }

    const onHand = new Map<string, number>();

    for (const row of (inventory.data ?? []) as {
      product_id: string;
      quantity: number;
    }[]) {
      onHand.set(row.product_id, Number(row.quantity) || 0);
    }

    const sold = new Map<string, number>();

    for (const movement of (sales.data ?? []) as {
      product_id: string;
      quantity: number;
    }[]) {
      sold.set(
        movement.product_id,
        (sold.get(movement.product_id) ?? 0) +
          Math.abs(Number(movement.quantity) || 0)
      );
    }

    const incoming = new Map<string, number>();

    for (const order of (openOrders.data ?? []) as any[]) {
      for (const item of (order.order_items ?? []) as {
        product_id: string;
        quantity: number;
      }[]) {
        incoming.set(
          item.product_id,
          (incoming.get(item.product_id) ?? 0) +
            (Number(item.quantity) || 0)
        );
      }
    }

    const list: Orderable[] = ((assigned.data ?? []) as any[])
      .map((row) => row.product)
      .filter((product: any) => product && product.active)
      .map((product: any) => {
        const have = onHand.get(product.id) ?? 0;
        const inc = incoming.get(product.id) ?? 0;
        const sold30 = sold.get(product.id) ?? 0;
        const result = runway(have, sold30, WINDOW_DAYS, inc);

        return {
          id: product.id,
          name: product.name,
        sku: product.sku,
        barcode: product.barcode ?? null,
        category: product.category ?? null,
          onHand: have,
          incoming: inc,
          sold30,
          dailyRate: result.dailyRate,
          daysLeft: result.daysLeft,
          urgency: result.urgency,
          lowData: result.lowData,
          suggest: result.suggest,
        };
      })
      .sort(
        (a: Orderable, b: Orderable) =>
          Number(b.suggest > 0) - Number(a.suggest > 0) ||
          RANK[a.urgency] - RANK[b.urgency] ||
          (a.daysLeft ?? Infinity) - (b.daysLeft ?? Infinity) ||
          a.name.localeCompare(b.name)
      );

    this.products.set(list);
    this.productsLoading.set(false);
  }

  onSearchInput(value: string): void {
    this.search.set(value);
    this.scanFeedback.set(null);
  }

  onSearchEnter(event: Event): void {
    event.preventDefault();

    const raw = this.search().trim();
    if (!raw) return;

    const value = raw.toLowerCase();
    const exactMatches = this.products().filter((product) =>
      product.sku.trim().toLowerCase() === value ||
      product.barcode?.trim().toLowerCase() === value
    );

    if (exactMatches.length === 0) {
      this.scanFeedback.set({
        ok: false,
        message: `No assigned product found for "${raw}".`,
      });
      this.focusSearch();
      return;
    }

    if (exactMatches.length > 1) {
      this.scanFeedback.set({
        ok: false,
        message: `More than one product uses "${raw}". Check the product catalogue.`,
      });
      this.focusSearch();
      return;
    }

    const product = exactMatches[0];
    this.setQty(product.id, this.qty(product.id) + 1);
    this.search.set('');
    this.onlyNeeded.set(false);
    this.scanFeedback.set({
      ok: true,
      message: `${product.name} added. Quantity: ${this.qty(product.id)}.`,
    });
    this.focusSearch();
  }

  daysText(
    onHandValue: number,
    days: number | null,
    sold = MIN_SALES
  ): string {
    return runwayLabel(
      onHandValue,
      days,
      sold > 0 && sold < MIN_SALES
    );
  }

  afterTone(days: number | null): string {
    if (days === null) {
      return 'bg-slate-100 text-slate-600 ring-slate-200';
    }

    if (days <= CRITICAL_DAYS) {
      return 'bg-red-50 text-red-700 ring-red-200';
    }

    if (days <= WARNING_DAYS) {
      return 'bg-amber-50 text-amber-700 ring-amber-200';
    }

    if (days > OVERSTOCK_DAYS) {
      return 'bg-amber-50 text-amber-700 ring-amber-200';
    }

    return 'bg-green-50 text-green-700 ring-green-200';
  }

  isOverstock(days: number | null): boolean {
    return days !== null && days > OVERSTOCK_DAYS;
  }

  coveredByIncoming(product: Orderable): boolean {
    return (
      product.incoming > 0 &&
      product.suggest === 0 &&
      (
        product.urgency === 'out' ||
        product.urgency === 'critical' ||
        product.urgency === 'warning'
      )
    );
  }

  targetQty(line: BasketLine): number {
    return Math.max(
      0,
      Math.ceil(
        line.dailyRate * TARGET_DAYS -
          Math.max(0, line.onHand) -
          line.incoming
      )
    );
  }

  delay(index: number): number {
    return Math.min(index, 12) * 30;
  }

  qty(id: string): number {
    return this.cart()[id] ?? 0;
  }

  private setQty(id: string, qty: number): void {
    const next = Math.max(
      0,
      Math.min(9999, Math.floor(qty) || 0)
    );

    this.cart.update((cart) => ({
      ...cart,
      [id]: next,
    }));

    this.error.set(null);
    this.bumpId.set(null);

    setTimeout(() => {
      this.bumpId.set(id);
    });
  }

  add(product: Orderable, delta = 1): void {
    this.setQty(product.id, this.qty(product.id) + delta);
  }

  addSuggested(product: Orderable): void {
    if (product.suggest > 0) {
      this.setQty(product.id, product.suggest);
    }
  }

  addAllSuggested(): void {
    const next = { ...this.cart() };

    for (const product of this.suggestedProducts()) {
      next[product.id] = Math.max(
        next[product.id] ?? 0,
        product.suggest
      );
    }

    this.cart.set(next);
    this.error.set(null);
  }

  onQtyInput(id: string, value: string): void {
    this.setQty(id, Number(value));
  }

  remove(id: string): void {
    this.cart.update((cart) => ({
      ...cart,
      [id]: 0,
    }));
  }

  clear(): void {
    this.cart.set({});
  }

  @HostListener('document:keydown', ['$event'])
  onKey(event: KeyboardEvent): void {
    if (
      (event.ctrlKey || event.metaKey) &&
      event.key === 'Enter' &&
      !this.placed() &&
      !!this.selectedSupplierId() &&
      this.totalUnits() > 0 &&
      !this.saving()
    ) {
      event.preventDefault();
      void this.submit();
    }
  }

  async submit(): Promise<void> {
    const objectId = this.selectedObjectId();
    const supplierId = this.selectedSupplierId();
    const snapshot = this.lines();

    if (!objectId || snapshot.length === 0 || this.saving()) {
      return;
    }

    if (!supplierId) {
      this.error.set('Select a supplier before placing the order.');
      return;
    }

    this.saving.set(true);
    this.error.set(null);

    const result = await this.orders.create(
      objectId,
      supplierId,
      this.comment().trim() || null,
      snapshot.map((line) => ({
        product_id: line.id,
        quantity: line.qty,
      }))
    );

    this.saving.set(false);

    if (result.error || !result.order) {
      this.error.set(
        result.error ?? 'Could not create the order.'
      );
      return;
    }

    this.placed.set({
      orderId: result.order.id,
      orderNumber: result.order.orderNumber,
      objectName: this.objectName(),
      supplierName: this.supplierName(),
      units: snapshot.reduce(
        (sum, line) => sum + line.qty,
        0
      ),
      lines: snapshot,
    });

    window.scrollTo({
      top: 0,
      behavior: 'smooth',
    });
  }

  async newOrder(): Promise<void> {
    this.placed.set(null);
    this.cart.set({});
    this.comment.set('');
    this.search.set('');
    this.scanFeedback.set(null);
    this.onlyNeeded.set(false);
    this.selectedSupplierId.set(null);
    this.error.set(null);

    const objectId = this.selectedObjectId();

    if (objectId) {
      await this.loadProducts(objectId);
    }
  }
}

function partyBlock(
  label: string,
  party: NonNullable<Order['company_snapshot']>,
  email: string | null | undefined
): string {
  const cityLine = [party.postal_code, party.city].filter(Boolean).join(' ');
  return `<article class="party">
    <div class="label">${esc(label)}</div>
    <div class="party-name">${esc(party.name ?? '')}</div>
    ${line(party.address)}
    ${line(cityLine)}
    ${line(party.country)}
    ${party.tax_id ? `<div class="muted">Tax ID: ${esc(party.tax_id)}</div>` : ''}
    ${party.registration_number ? `<div class="muted">Reg. no: ${esc(party.registration_number)}</div>` : ''}
    ${email ? `<div class="muted">${esc(email)}</div>` : ''}
    ${party.phone ? `<div class="muted">${esc(party.phone)}</div>` : ''}
  </article>`;
}

function line(value: string | null | undefined): string {
  return value ? `<div>${esc(value)}</div>` : '';
}

function esc(value: unknown): string {
  return String(value ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

function formatDate(iso: string): string {
  return new Intl.DateTimeFormat('en-GB', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
  }).format(new Date(iso));
}

  /**
   * RLS still controls which orders the current user
   * is allowed to count.
   */
  private async loadCounts(): Promise<void> {
    const objectId = this.selectedObjectId();

    const results = await Promise.all(
      this.statuses.map((status) => {
        let query = supabase
          .from('orders')
          .select('id', {
            count: 'exact',
            head: true,
          })
          .eq('status', status);

        if (objectId) {
          query = query.eq(
            'object_id',
            objectId
          );
        }

        return query;
      })
    );

    const next: Record<OrderStatus, number> = {
      PENDING: 0,
      APPROVED: 0,
      DELIVERED: 0,
      REJECTED: 0,
    };

    results.forEach((result, index) => {
      next[this.statuses[index]] =
        result.count ?? 0;
    });

    this.counts.set(next);

    let exceptionQuery = supabase
      .from('order_fulfillment')
      .select('order_id')
      .neq('variance', 0);

    const {
      data: exceptionRows,
      error: exceptionError,
    } = await exceptionQuery;

    if (exceptionError) {
      this.exceptionCount.set(0);
      return;
    }

    const exceptionIds = [
      ...new Set(
        (exceptionRows ?? []).map(
          (row) => row.order_id as string
        )
      ),
    ];

    if (
      !objectId ||
      exceptionIds.length === 0
    ) {
      this.exceptionCount.set(
        objectId
          ? 0
          : exceptionIds.length
      );

      return;
    }

    const {
      count: locationExceptionCount,
    } = await supabase
      .from('orders')
      .select('id', {
        count: 'exact',
        head: true,
      })
      .eq('status', 'DELIVERED')
      .eq('object_id', objectId)
      .in('id', exceptionIds);

    this.exceptionCount.set(
      locationExceptionCount ?? 0
    );
  }

  itemCount(order: Order): number {

        0
      ) ?? 0
    );

  /**
   * Where the order currently sits in:
   *
   * PENDING → APPROVED → DELIVERED
   */
  stepState(
    order: Order,
    step: number
  ): StepState {
    if (order.status === 'REJECTED') {
      if (step === 0) {
        return 'done';
      }

      if (step === 1) {
        return 'rejected';
      }

      return 'todo';
    }

    const index = this.flow.indexOf(
      order.status
    );

    if (step < index) {
      return 'done';
    }

    if (step === index) {
      return order.status === 'DELIVERED'
        ? 'done'
        : 'current';
    }

    return 'todo';
  }

  stepDot(
    order: Order,
    step: number
  ): string {
    const state = this.stepState(
      order,
      step
    );

    if (state === 'rejected') {
      return 'bg-status-rejected';
    }

    if (state === 'todo') {
      return 'bg-slate-200';
    }

    return this.meta[
      this.flow[step]
    ].dot;
  }

  /**
   * Connector leading INTO a lifecycle step.
   */
  stepLine(
    order: Order,
    step: number
  ): string {
    const state = this.stepState(
      order,
      step
    );

    if (state === 'rejected') {
      return 'bg-red-200';
    }

    if (state === 'todo') {
      return 'bg-slate-200';
    }

    return this.meta[
      this.flow[step]
    ].dot;
  }

  nextHint(order: Order): string {
    if (order.status === 'PENDING') {
      return this.auth.isAdmin()
        ? 'Needs your approval'
        : 'Waiting for approval';
    }

    if (order.status === 'APPROVED') {
      return 'Ready to receive';
    }

    return this.meta[
      order.status
    ].hint;
  }

  ago(iso: string): string {
    const seconds = Math.max(
      0,
      (
        Date.now() -
        new Date(iso).getTime()
      ) / 1000
    );

    if (seconds < 60) {
      return 'just now';
    }

    if (seconds < 3600) {
      return `${Math.floor(
        seconds / 60
      )}m ago`;
    }

    if (seconds < 86400) {
      return `${Math.floor(
        seconds / 3600
      )}h ago`;
    }

    if (seconds < 604800) {
      return `${Math.floor(
        seconds / 86400
      )}d ago`;
    }

    return new Date(
      iso
    ).toLocaleDateString(
      'en-GB',
      {
        day: 'numeric',
        month: 'short',
      }
    );
  }

  delay(index: number): number {
    return Math.min(
      index,
      12
    ) * 35;
  }

  variance(order: Order) {
    return this.deliveries
      .variances()[order.id];
  }

  varianceLabel(
    order: Order
  ): string {
    const variance =
      this.variance(order);

    if (!variance) {
      return '';
    }

    const parts: string[] = [];

    if (variance.short > 0) {
      parts.push(
        `${variance.short} short`
      );
    }

    if (variance.over > 0) {
      parts.push(
        `${variance.over} over`
      );
    }

    return parts.join(' · ');
  }

    this.flashId.set(id);

    setTimeout(
      () =>
        this.flashId.set(null),
      2300
    );
  }

  async approve(
    order: Order
  ): Promise<void> {
    const error =
      await this.svc.setStatus(
        order.id,
        'APPROVED'
      );

    if (error) {
      this.toast.error(error);
      return;
    }

    this.reload(
      this.svc.page()
    );

    this.flash(order.id);

    this.toast.success(
      `${order.order_number} approved`,
      {
        action: {
          label: 'Undo',

          run: async () => {
            const undoError =
              await this.svc.setStatus(
                order.id,
                'PENDING'
              );

            if (undoError) {
              this.toast.error(
                undoError
              );

              return;
            }

            this.reload(
              this.svc.page()
            );

            this.flash(
              order.id
            );

            this.toast.info(
              'Approval undone — back to Pending.'
            );
          },
        },
      }
    );
  }

  async reject(
    order: Order
  ): Promise<void> {
    const ok =
      await this.confirm.ask({
        title:
          'Reject this order?',

        message:
          `${order.order_number} for ` +
          `${order.object?.name ?? 'this location'} ` +
          `will be marked REJECTED. ` +
          `This can't be undone.`,

        confirmLabel:
          'Reject order',

        tone:
          'danger',
      });

    if (!ok) {
      return;
    }

    const error =
      await this.svc.setStatus(
        order.id,
        'REJECTED'
      );

    if (error) {
      this.toast.error(error);
      return;
    }

    this.reload(
      this.svc.page()
    );

    this.flash(
      order.id
    );

    this.toast.info(
      `${order.order_number} rejected.`
    );
  }

order.service.ts
import { Injectable, computed, signal } from '@angular/core';
import { supabase } from '../../core/supabase/supabase.client';
import { Order, OrderStatus, NewOrderLine } from './order.model';
import { demoFriendly } from '../../core/demo-error';

export interface OrdersQuery {
  page: number;
  pageSize: number;
  status: OrderStatus | '';
  objectId: string;
  search: string;
  exceptionsOnly: boolean;
}

export interface CreatedOrder {
  id: string;
  orderNumber: string;
}

const BLOCKED = 'This action is disabled in the demo account.';

@Injectable({ providedIn: 'root' })
export class OrdersService {
  readonly orders = signal<Order[]>([]);
  readonly loading = signal(false);
  readonly error = signal<string | null>(null);
  readonly page = signal(1);
  readonly pageSize = signal(10);
  readonly total = signal(0);
  readonly totalPages = computed(() => Math.max(1, Math.ceil(this.total() / this.pageSize())));

  async load(params: Partial<OrdersQuery> = {}): Promise<void> {
  const page = params.page ?? this.page();
  const pageSize = params.pageSize ?? this.pageSize();
  const status = params.status ?? '';
  const objectId = params.objectId ?? '';
  const search = (params.search ?? '').trim();
  const exceptionsOnly = params.exceptionsOnly ?? false;

  const from = (page - 1) * pageSize;
  const to = from + pageSize - 1;

  this.loading.set(true);
  this.error.set(null);

  let exceptionOrderIds: string[] = [];

  if (exceptionsOnly) {
    const { data: exceptionRows, error: exceptionError } =
      await supabase
        .from('order_fulfillment')
        .select('order_id')
        .neq('variance', 0);

    if (exceptionError) {
      this.error.set(exceptionError.message);
      this.orders.set([]);
      this.total.set(0);
      this.page.set(1);
      this.loading.set(false);
      return;
    }

    exceptionOrderIds = [
      ...new Set(
        (exceptionRows ?? []).map(
          (row) => row.order_id as string
        )
      ),
    ];

    if (exceptionOrderIds.length === 0) {
      this.orders.set([]);
      this.total.set(0);
      this.page.set(1);
      this.pageSize.set(pageSize);
      this.loading.set(false);
      return;
    }
  }

  let query = supabase
    .from('orders')
    .select(`
      id,
      order_number,
      object_id,
      supplier_id,
      status,
      comment,
      created_by,
      created_at,
      approved_at,
      approved_by,
      company_snapshot,
      supplier_snapshot,
      delivery_snapshot,
      prepared_by_snapshot,
      object:objects (
        name
      ),
      supplier:suppliers (
        id,
        name,
        order_email,
        active
      ),
      order_items (
        id,
        order_id,
        product_id,
        quantity,
        product:products (
          name,
          sku
        )
      )
    `, {
      count: 'exact',
    })
    .order('created_at', {
      ascending: false,
    });

  if (exceptionsOnly) {
    query = query
      .eq('status', 'DELIVERED')
      .in('id', exceptionOrderIds);
  } else if (status) {
    query = query.eq(
      'status',
      status
    );
  }

  if (objectId) {
    query = query.eq(
      'object_id',
      objectId
    );
  }

  if (search) {
    const escaped = search.replace(
      /[%_,()]/g,
      ''
    );

    query = query.or(
      `order_number.ilike.%${escaped}%,comment.ilike.%${escaped}%`
    );
  }

  query = query.range(from, to);

  const {
    data,
    error,
    count,
  } = await query;

  if (error) {
    this.error.set(error.message);
    this.orders.set([]);
    this.total.set(0);
    this.loading.set(false);
    return;
  }

  this.orders.set(
    (data ?? []) as unknown as Order[]
  );

  this.page.set(page);
  this.pageSize.set(pageSize);
  this.total.set(count ?? 0);
  this.loading.set(false);
}

  async create(
    objectId: string,
    supplierId: string,
    comment: string | null,
    lines: NewOrderLine[]
  ): Promise<{ order: CreatedOrder | null; error: string | null }> {
    const { data: orderId, error } = await supabase.rpc('create_order_with_items', {
      p_object_id: objectId,
      p_comment: comment,
      p_items: lines.map((l) => ({ product_id: l.product_id, quantity: l.quantity })),
      p_supplier_id: supplierId,
    });

    if (error) return { order: null, error: friendly(error.message) };
    if (!orderId) return { order: null, error: 'The order was created but its ID was not returned.' };

    const { data: created, error: readError } = await supabase
      .from('orders')
      .select('id, order_number')
      .eq('id', orderId)
      .single();

    if (readError) return { order: null, error: friendly(readError.message) };
    return {
      order: { id: created.id, orderNumber: created.order_number },
      error: null,
    };
  }

  async setStatus(orderId: string, status: OrderStatus): Promise<string | null> {
    const { data, error } = await supabase
      .from('orders')
      .update({ status })
      .eq('id', orderId)
      .select('id');

    if (error) return friendly(error.message);
    if (!data || data.length === 0) return BLOCKED;
    return null;
  }

  async next(params: Omit<Partial<OrdersQuery>, 'page'> = {}): Promise<void> {
    if (this.page() >= this.totalPages()) return;
    await this.load({ ...params, page: this.page() + 1 });
  }

  async previous(params: Omit<Partial<OrdersQuery>, 'page'> = {}): Promise<void> {
    if (this.page() <= 1) return;
    await this.load({ ...params, page: this.page() - 1 });
  }
}

function friendly(message: string): string {
  const demo = demoFriendly(message);
  if (demo !== message) return demo;
  if (message.includes('Select a supplier')) return 'Select a supplier before approving this order.';
  if (message.includes('Inactive supplier')) return 'This supplier is inactive. Choose an active supplier.';
  if (message.includes('Complete Settings')) return 'Complete Settings → Company profile before approving purchase orders.';
  if (message.includes('Supplier does not exist')) return 'The selected supplier is no longer available.';
  return message;
}
