export class OrdersComponent implements OnInit {

  readonly deliveries = inject(DeliveriesService);

  private toast = inject(ToastService);
  private confirm = inject(ConfirmService);
  private route = inject(ActivatedRoute);
  private router = inject(Router);

  // =========================================================
  // Filters
  // =========================================================

  readonly objects = signal<BusinessObject[]>([]);

  readonly selectedObjectId = signal<string>('');

  readonly filterLocations = computed(() => [
    {
      id: '',
      name: 'All locations',
    },
    ...this.objects(),
  ]);

  readonly statusFilter = signal<OrderStatus | ''>('');
  readonly exceptionsOnly = signal(false);
  readonly exceptionCount = signal(0);
  readonly search = signal('');

  readonly statuses: OrderStatus[] = [
    'PENDING',
    'APPROVED',
    'DELIVERED',
    'REJECTED',
  ];

  /**
   * Normal fulfillment path.
   *
   * REJECTED is intentionally not part of this flow.
   */
  readonly flow: OrderStatus[] = [
    'PENDING',
    'APPROVED',
    'DELIVERED',
  ];

  readonly counts = signal<Record<OrderStatus, number>>({
    PENDING: 0,
    APPROVED: 0,
    DELIVERED: 0,
    REJECTED: 0,
  });

  readonly countsTotal = computed(() =>
    Object.values(this.counts()).reduce(
      (sum, value) => sum + value,
      0
    )
  );

  /**
   * Recently changed order.
   * Used for the flash animation after status updates.
   */
  readonly flashId = signal<string | null>(null);

  private searchTimer: ReturnType<typeof setTimeout> | null = null;

  // =========================================================
  // Confirm delivery drawer
  // =========================================================

  readonly deliveringOrder = signal<Order | null>(null);
  readonly deliverDrawerOpen = signal(false);

  private lastDeliveringId: string | null = null;

  // =========================================================
  // Delivery summary drawer
  // =========================================================

  readonly summaryOrder = signal<Order | null>(null);
  readonly summaryDrawerOpen = signal(false);

  // =========================================================
  // Status presentation
  // =========================================================

  readonly meta: Record<OrderStatus, StatusMeta> = {

  constructor() {
    /**
     * Load variance information only for delivered orders
     * currently visible on the page.
     */
    effect(() => {
      const deliveredIds = this.svc
        .orders()
        .filter((order) => order.status === 'DELIVERED')
        .map((order) => order.id);

      void this.deliveries.loadVariances(deliveredIds);
    });
  }

  // =========================================================
  // Init
  // =========================================================

  async ngOnInit(): Promise<void> {
    if (this.auth.isAdmin()) {
      await this.loadObjects();
    }

    const focusId = this.route.snapshot.queryParamMap.get('focus');
    if (focusId) {
      await this.focusOrder(focusId);
    } else {
      this.reload(1);
    }
  }

  /** Finds the exact order regardless of pagination, then scrolls and flashes it. */
  private async focusOrder(orderId: string): Promise<void> {
    const { data: target, error } = await supabase
      .from('orders')
      .select('id, order_number, object_id, status')
      .eq('id', orderId)
      .maybeSingle();

    if (error || !target) {
      this.reload(1);
      return;
    }

    this.exceptionsOnly.set(false);
    this.statusFilter.set(target.status as OrderStatus);
    this.selectedObjectId.set(target.object_id);
    this.search.set(target.order_number);

    await Promise.all([
      this.svc.load({
        page: 1,
        status: this.statusFilter(),
        objectId: this.selectedObjectId(),
        search: this.search(),
        exceptionsOnly: false,
      }),
      this.loadCounts(),
    ]);

    this.scrollAndFlashOrder(orderId);

    void this.router.navigate([], {
      relativeTo: this.route,
      queryParams: { focus: null },
      queryParamsHandling: 'merge',
      replaceUrl: true,
    });
  }

  private scrollAndFlashOrder(orderId: string): void {
    this.flashId.set(orderId);

    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const nodes = Array.from(
          document.querySelectorAll<HTMLElement>(`[data-order-id="${orderId}"]`)
        );
        const target = nodes.find((node) => node.offsetParent !== null) ?? nodes[0];
        target?.scrollIntoView({ behavior: 'smooth', block: 'center' });
      });
    });

    setTimeout(() => this.flashId.set(null), 2300);
  }

  private async loadObjects(): Promise<void> {
    const { data } = await supabase
      .from('objects')
      .select('*')
      .eq('active', true)
      .order('name');

    this.objects.set(
      (data ?? []) as BusinessObject[]
    );
  }

  // =========================================================
  // Counts
  // =========================================================

  /**
   * RLS still controls which orders the current user
   * is allowed to count.
   */
  private async loadCounts(): Promise<void> {
    const objectId = this.selectedObjectId();

    const results = await Promise.all(
      this.statuses.map((status) => {
        let query = supabase
          .from('orders')
          .select('id', {
            count: 'exact',
            head: true,
          })
          .eq('status', status);

        if (objectId) {
          query = query.eq(
            'object_id',
            objectId
          );
        }

        return query;
      })
    );

    const next: Record<OrderStatus, number> = {
      PENDING: 0,
      APPROVED: 0,
      DELIVERED: 0,
      REJECTED: 0,
    };

    results.forEach((result, index) => {
      next[this.statuses[index]] =
        result.count ?? 0;
    });

    this.counts.set(next);

    // "Needs resolution" = open discrepancy LINES (not orders)
    const { data: openRows, error: openError } = await supabase.rpc(
      'delivery_exceptions',
      { p_status: 'open', p_object_id: objectId || null, p_delivery_id: null }
    );

    this.exceptionCount.set(openError ? 0 : (openRows ?? []).length);
  }

  // =========================================================
  // Loading / filters
  // =========================================================

  reload(page = 1): void {
    void this.svc.load({
      page,
      status: this.statusFilter(),
      objectId: this.selectedObjectId(),
      search: this.search(),
      exceptionsOnly: this.exceptionsOnly(),
    });

    void this.loadCounts();
  }

  onStatusChange(status: OrderStatus | ''): void {
    this.exceptionsOnly.set(false);

    this.statusFilter.set(
      this.statusFilter() === status
        ? ''
        : status
    );

    this.reload(1);
  }

  showAll(): void {
    this.statusFilter.set('');
    this.exceptionsOnly.set(false);
    this.reload(1);
  }

  showExceptions(): void {
    const willEnable =
      !this.exceptionsOnly();

    this.exceptionsOnly.set(
      willEnable
    );

    this.statusFilter.set('');

    this.reload(1);
  }

  onObjectChange(
    objectId: string
  ): void {
    this.selectedObjectId.set(objectId);
    this.reload(1);
  }

  onSearch(value: string): void {
    this.search.set(value);

    if (this.searchTimer) {
      clearTimeout(this.searchTimer);
    }

    this.searchTimer = setTimeout(
      () => this.reload(1),
      250
    );
  }

  previous(): void {
    void this.svc.previous({
      status: this.statusFilter(),
      objectId: this.selectedObjectId(),
      search: this.search(),
      exceptionsOnly: this.exceptionsOnly(),
    });
  }

  next(): void {
    void this.svc.next({
      status: this.statusFilter(),
      objectId: this.selectedObjectId(),
      search: this.search(),
      exceptionsOnly: this.exceptionsOnly(),
    });
  }

  itemCount(order: Order): number {
    return (
      order.order_items?.reduce(
        (sum, item) =>
          sum + item.quantity,
        0
      ) ?? 0
    );
  }

  /**
   * Where the order currently sits in:
   *
   * PENDING → APPROVED → DELIVERED
   */
  stepState(
    order: Order,
    step: number
  ): StepState {
    if (order.status === 'REJECTED') {
      if (step === 0) {
        return 'done';
      }

      if (step === 1) {
        return 'rejected';
      }

      return 'todo';
    }

    const index = this.flow.indexOf(
      order.status
    );

    if (step < index) {
      return 'done';
    }

    if (step === index) {
      return order.status === 'DELIVERED'
        ? 'done'
        : 'current';
    }

    return 'todo';
  }

  stepDot(
    order: Order,
    step: number
  ): string {
    const state = this.stepState(
      order,
      step
    );

    if (state === 'rejected') {
      return 'bg-status-rejected';
    }

    if (state === 'todo') {
      return 'bg-slate-200';
    }

    return this.meta[
      this.flow[step]
    ].dot;
  }

  /**
   * Connector leading INTO a lifecycle step.
   */
  stepLine(
    order: Order,
    step: number
  ): string {
    const state = this.stepState(
      order,
      step
    );

    if (state === 'rejected') {
      return 'bg-red-200';
    }

    if (state === 'todo') {
      return 'bg-slate-200';
    }

    return this.meta[
      this.flow[step]
    ].dot;
  }

  nextHint(order: Order): string {
    if (order.status === 'PENDING') {
      return this.auth.isAdmin()
        ? 'Needs your approval'
        : 'Waiting for approval';
    }

    if (order.status === 'APPROVED') {
      return 'Ready to receive';
    }

    return this.meta[
      order.status
    ].hint;
  }

  ago(iso: string): string {
    const seconds = Math.max(
      0,
      (
        Date.now() -
        new Date(iso).getTime()
      ) / 1000
    );

    if (seconds < 60) {
      return 'just now';
    }

    if (seconds < 3600) {
      return `${Math.floor(
        seconds / 60
      )}m ago`;
    }

    if (seconds < 86400) {
      return `${Math.floor(
        seconds / 3600
      )}h ago`;
    }

    if (seconds < 604800) {
      return `${Math.floor(
        seconds / 86400
      )}d ago`;
    }

    return new Date(
      iso
    ).toLocaleDateString(
      'en-GB',
      {
        day: 'numeric',
        month: 'short',
      }
    );
  }

  delay(index: number): number {
    return Math.min(
      index,
      12
    ) * 35;
  }

  variance(order: Order) {
    return this.deliveries
      .variances()[order.id];
  }

  varianceLabel(
    order: Order
  ): string {
    const variance =
      this.variance(order);

    if (!variance) {
      return '';
    }

    const parts: string[] = [];

    if (variance.short > 0) {
      parts.push(
        `${variance.short} short`
      );
    }

    if (variance.over > 0) {
      parts.push(
        `${variance.over} over`
      );
    }

    const label = parts.join(' · ');
    return variance.open > 0
      ? `${label} · ${variance.open} to resolve`
      : `${label} · resolved`;
  }

  private flash(id: string): void {
    this.flashId.set(id);

    setTimeout(
      () =>
        this.flashId.set(null),
      2300
    );
  }

  // =========================================================
  // Actions
  // =========================================================
  printOrder(order: Order): void {
    const error = printPurchaseOrder(order);

    if (error) {
      this.toast.error(error);
    }
  }


  async approve(
    order: Order
  ): Promise<void> {
    const error =
      await this.svc.setStatus(
        order.id,
        'APPROVED'
      );

    if (error) {
      this.toast.error(error);
      return;
    }

    this.reload(
      this.svc.page()
    );

    this.flash(order.id);

    this.toast.success(
      `${order.order_number} approved`,
      {
        action: {
          label: 'Undo',

          run: async () => {
            const undoError =
              await this.svc.setStatus(
                order.id,
                'PENDING'
              );

            if (undoError) {
              this.toast.error(
                undoError
              );

              return;
            }

            this.reload(
              this.svc.page()
            );

            this.flash(
              order.id
            );

            this.toast.info(
              'Approval undone — back to Pending.'
            );
          },
        },
      }
    );
  }

  async reject(
    order: Order
  ): Promise<void> {
    const ok =
      await this.confirm.ask({
        title:
          'Reject this order?',

        message:
          `${order.order_number} for ` +
          `${order.object?.name ?? 'this location'} ` +
          `will be marked REJECTED. ` +
          `This can't be undone.`,

        confirmLabel:
          'Reject order',

        tone:
          'danger',
      });

    if (!ok) {
      return;
    }

    const error =
      await this.svc.setStatus(
        order.id,
        'REJECTED'
      );

    if (error) {
      this.toast.error(error);
      return;
    }

    this.reload(
      this.svc.page()
    );

    this.flash(
      order.id
    );

    this.toast.info(
      `${order.order_number} rejected.`
    );
  }

  onDelivered(
    summary?: string
  ): void {
    this.reload(
      this.svc.page()

    this.toast.success(
      summary ??
      'Delivery recorded — stock updated.'
    );
  }

      order

    this.summaryDrawerOpen.set(

    );
  }

  /** A discrepancy was resolved in the receipt drawer: refresh list + counter. */
  onResolved(): void {
    this.reload(this.svc.page());
  }

  closeSummary(): void {
    this.summaryDrawerOpen.set(
      false
    );

    this.summaryOrder.set(
      null
    );
  }
